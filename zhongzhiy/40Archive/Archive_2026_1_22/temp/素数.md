素数除了$2, 3$都是$6k+1$
优化判素数
```cpp fold
bool isprimer(int x) {
    if(x <= 1) return 0;
    if(x == 2 || x == 3) return 1;
    if(x % 2 == 0 || x % 3 == 0) return 0;

	//遍历所有的6+-1的数
    for(int i = 5;i * i <= x; i += 6) {
        if(x % i == 0 || x % (i + 2) == 0) return 0;
    }
    return 1;
}

```
