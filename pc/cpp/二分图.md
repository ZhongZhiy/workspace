void dfs(int x, int fa) {
    if(!is) return;
    for(auto c : adj[x]) {
        if(c == fa) continue;

        if(color[c] == color[x]) {
            is = false;
            return;
        }else if(color[c] == 3 - color[x]) continue;

        color[c] = 3 - color[x];
        dfs(c, x);
    }
}
