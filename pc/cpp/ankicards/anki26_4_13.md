# 算法

## [天梯地图](https://pintia.cn/problem-sets/994805046380707840/exam/problems/type/7?problemSetProblemId=994805051153825792&page=1)
给 $n$ 个节点, $m$ 条边, 有的边是双向边, 有的是单向边, 每条边有距离和通行花费时间, 
求从起点到终点的最小距离和最小时间, 如果有相同的最小距离就输出通过节点数最少的, 如果有相同最小时间, 就输出距离最小的
$(2 <= N <= 500) $ 节点为 $(0, n-1)$
---
题解:
这是两个双标尺最短路, 比较多的细节需要处理
首先, 确保在dijstra最后是需要的结果, 因此在遍历边的时候就需要更新处理最小时间和第二指标, 
此外, 在更新的时候还需要记录父节点, 方便最后输出
最后, 分拆成两个函数写会比较清晰
代码:
```cpp

constexpr int N = 505;
struct E{
    int u, l, t;
    E(){}
    E(int u, int l, int t):u(u), l(l), t(t){}
};

vector<E> e[N];
int st, ed, mn_time, mn_dis;
struct Node{
    int id, csm;
    Node(){}
    Node(int id, int csm):id(id), csm(csm){}
    bool operator <(const Node &a)const{
        return csm > a.csm;
    }
};
int n, m;
pair<int, vi>  get_least_time(){
    priority_queue<Node> q;
    vi lt(n + 1, INT_MAX), ld(n + 1, INT_MAX), pref(n + 1, -1);
    q.push({st, 0});
    lt[st] = ld[st] = 0;

    while(!q.empty()) {
        auto [id, csm] = q.top(); q.pop();
        if(csm > lt[id]) continue;

        // continue;
        for(auto &[u, l, t] : e[id]) {
            if(t + csm < lt[u]) {
                lt[u] = t + csm;
                ld[u] = ld[id] + l;
                pref[u] = id;
                q.push({u, lt[u]});
            }else if(t + csm == lt[u] && l + ld[id] < ld[u]) {
                ld[u] = ld[id] + l;
                pref[u] = id;
            }
        }
    }

    vi least_t;
    int tp = ed;
    while(pref[tp] != -1) {
        least_t.push_back(tp);
        tp = pref[tp];
    }least_t.push_back(st);
    reverse(all(least_t));
    return {lt[ed], least_t};
}

pair<int, vi> get_least_dis() {
    vi ld(n + 1, INT_MAX), ln(n + 1, INT_MAX), pref(n + 1, -1);
    priority_queue<Node> q;
    q.push({st, 0});
    ld[st] = 0; ln[st] = 0;

    while(!q.empty()) {
        auto [id, d] = q.top(); q.pop();
        if(ld[id] < d) continue;

        for(auto &[u, l, t] : e[id]) {
            if(d + l < ld[u]) {
                ld[u] = d + l;
                ln[u]  = ln[id] + 1;
                pref[u] = id;
                q.push({u, ld[u]});
            }else if(d + l == ld[u] && ln[u] > ln[id] + 1) {
                ln[u] = ln[id] + 1;
                pref[u] = id;
            }
        }
    }

    vi least_d;
    int tp = ed;
    while(pref[tp] != -1) {
        least_d.push_back(tp);
        tp = pref[tp];
    }
    least_d.push_back(st);
    reverse(all(least_d));
    return {ld[ed], least_d};
}

signed main() {
    hello;
     cin >> n >> m;
    fi(m) {
        int u, v, o, l, t; cin >> u >> v >> o >> l >> t;
        e[u].emplace_back(v, l, t);
        if(!o) e[v].emplace_back(u, l, t);
    }
    cin >> st >> ed;

    auto [lt, least_t] = get_least_time();
    auto [ld, least_d] = get_least_dis();

    if(least_d.size() == least_t.size()) {
        bool same = 1;
        for(int i = 0;i < least_d.size(); ++i) {
            if(least_d[i] != least_t[i]){
                same = 0;
                break;
            }
        }
        if(same){
            cout << "Time = " << lt << "; Distance = " << ld << ": ";
            for(int i = 0;i < least_d.size(); ++i) cout << least_d[i] << (i != least_d.size() -1? " => " :"");
            return 0;
        }
    }
    cout << "Time = " << lt << ": "; for(int i = 0;i < least_t.size(); ++i) cout << least_t[i] << (i != least_t.size()-1 ? " => " :"");cout << endl;
    cout << "Distance = " << ld << ": "; for(int i = 0;i < least_d.size(); ++i) cout << least_d[i] << (i != least_d.size()-1 ? " => " :"");

    world;
}
```
